CREATE TABLE clientes (
    id_cliente NUMBER PRIMARY KEY,
    nombre VARCHAR2(100) NOT NULL,
    correo VARCHAR2(100) UNIQUE,
    tipo_cliente VARCHAR2(20) DEFAULT 'REGULAR',
    CHECK (tipo_cliente IN ('REGULAR', 'VIP', 'EMPRESA'))

CREATE TABLE productos (
    id_producto NUMBER PRIMARY KEY,
    nombre VARCHAR2(100) NOT NULL,
    precio NUMBER(10,2) NOT NULL,
    stock NUMBER NOT NULL,
    CHECK (precio >= 0),
    CHECK (stock >= 0)
);

CREATE TABLE pedidos (
    id_pedido NUMBER PRIMARY KEY,
    id_cliente NUMBER NOT NULL,
    fecha DATE NOT NULL,
    total NUMBER(10,2) DEFAULT 0,
    estado VARCHAR2(20) DEFAULT 'PENDIENTE',
    FOREIGN KEY (id_cliente) REFERENCES clientes(id_cliente),
    CHECK (estado IN ('PENDIENTE', 'PROCESANDO', 'ENVIADO', 'ENTREGADO', 'CANCELADO')),
    CHECK (total >= 0)
);

CREATE TABLE detalles_pedidos (
    id_detalle NUMBER PRIMARY KEY,
    id_pedido NUMBER NOT NULL,
    id_producto NUMBER NOT NULL,
    cantidad NUMBER NOT NULL,
    precio_unitario NUMBER(10,2) NOT NULL,
    FOREIGN KEY (id_pedido) REFERENCES pedidos(id_pedido),
    FOREIGN KEY (id_producto) REFERENCES productos(id_producto),
    CHECK (cantidad > 0),
    CHECK (precio_unitario >= 0)
);

Módulo 1: Los 3 Ciclos Básicos (FOR, FOR REVERSE, WHILE)

Antes de meter las manos en las modificaciones de datos (DML), aprenderemos a usar ciclos para automatizar iteraciones lógicas básicas.

FOR

Iteraciones Determinadas

Se utiliza cuando sabemos **exactamente cuántas veces** queremos repetir un código (ej. del 1 al 10). La variable se incrementa automáticamente.

FOR REVERSE

Cuenta Regresiva

Hace lo mismo que el FOR, pero decrementa la variable desde el límite superior al inferior (ej. cuenta atrás de stock).

WHILE

Control por Condición

Se repite **mientras una condición de negocio sea verdadera**. Requiere modificar la variable manualmente para evitar bucles infinitos.

Ejercicio 1.1

Simulación de Cuotas Mensuales (Ciclo FOR)

OBJETIVO: Consultar el total del pedido 100 y usar un ciclo FOR (de 3 a 6) para imprimir el valor que pagaría el cliente si decidiera dividir la compra en 3, 4, 5 o 6 cuotas mensuales.

Pista: Declara variables para almacenar el total y el valor de la cuota. Obtén el total del pedido 100 ($95.000) con SELECT total INTO ... y luego usa FOR i IN 3..6 LOOP para dividir el total en cuotas. 

DECLARE
    v_total pedidos.total%TYPE;
    v_cuota NUMBER(10,2);
BEGIN
    -- Obtenemos el total del pedido 100
    SELECT total INTO v_total 
    FROM pedidos 
    WHERE id_pedido = 100;
    
    -- Imprimimos la simulación de cuotas
    FOR i IN 3..6 LOOP
        v_cuota := v_total / i;
        DBMS_OUTPUT.PUT_LINE('Alternativa ' || i || ' cuotas: $' || ROUND(v_cuota, 0) || ' mensual.');
    END LOOP;
END;
/
Ejercicio 1.2

Etiquetado Inverso de Bultos (Ciclo FOR REVERSE)

OBJETIVO: Consultar el stock del "Teclado Mecánico" (ID 20) y simular la impresión de etiquetas para su despacho en orden inverso (Bulto 3, Bulto 2, Bulto 1).

Pista: La sintaxis de reverse es FOR i IN REVERSE 1..v_stock LOOP. Esto te permite procesar o contar hacia atrás elementos que salen de bodega. 

DECLARE
    v_stock productos.stock%TYPE;
BEGIN
    SELECT stock INTO v_stock 
    FROM productos 
    WHERE id_producto = 20; -- Retorna stock = 3
    
    -- Cuenta regresiva: 3, 2, 1
    FOR i IN REVERSE 1..v_stock LOOP
        DBMS_OUTPUT.PUT_LINE('Imprimiendo etiqueta para bulto N° ' || i || ' de ' || v_stock);
    END LOOP;
END;
/
Ejercicio 1.3

Simulación de Liquidación por Caducidad (Ciclo WHILE)

OBJETIVO: El "Teclado Mecánico" (ID 20) no se vende. Consultar su precio ($45000) e imprimir cómo bajará su precio en un ciclo WHILE simulando una rebaja de $5000 por mes hasta llegar a un costo base de $25000.

Pista: Define variables para manejar el precio y un contador de meses. La condición del bucle será WHILE v_precio > 25000 LOOP. Recuerda restar 5000 en cada iteración para evitar bucles infinitos. 

DECLARE
    v_precio productos.precio%TYPE;
    v_mes NUMBER := 1;
BEGIN
    SELECT precio INTO v_precio 
    FROM productos 
    WHERE id_producto = 20;
    
    -- Rebajamos el precio hasta el costo base permitido
    WHILE v_precio > 25000 LOOP
        v_precio := v_precio - 5000;
        DBMS_OUTPUT.PUT_LINE('Mes ' || v_mes || ' de liquidación: Nuevo precio $' || v_precio);
        v_mes := v_mes + 1;
    END LOOP;
END;
/

Fase BINSERT dentro de FOR (Masivo)

Objetivo: Registrar 3 nuevos productos de prueba en la base de datos con IDs del 201 al 203, llamados 'Producto Demo 1', 'Producto Demo 2', y 'Producto Demo 3', con precio de 5000 y stock de 10.

DECLARE
    v_id_prod NUMBER;
    v_nombre VARCHAR2(100);
    v_contador NUMBER := 0;
BEGIN
    FOR i IN 1..3 LOOP
        v_id_prod := 200 + i;
        v_nombre := 'Producto Demo ' || i;
        
        INSERT INTO productos (id_producto, nombre, precio, stock)
        VALUES (v_id_prod, v_nombre, 5000, 10);
        
        v_contador := v_contador + 1;
    END LOOP;
    DBMS_OUTPUT.PUT_LINE('Total productos creados con FOR: ' || v_contador);
END;
/
Verificación con SELECT: SELECT * FROM productos WHERE id_producto BETWEEN 201 AND 203;

Fase CINSERT dentro de WHILE (Simulación de cola)

Objetivo: Registrar 3 pedidos simulados para el cliente 1 con IDs desde 204 al 206 usando un ciclo WHILE controlado por contador.

DECLARE
    v_id_ped NUMBER := 204;
    v_insertados NUMBER := 0;
BEGIN
    WHILE v_id_ped <= 206 LOOP
        INSERT INTO pedidos (id_pedido, id_cliente, fecha, total, estado)
        VALUES (v_id_ped, 1, SYSDATE, 0.00, 'PENDIENTE');
        
        v_insertados := v_insertados + 1;
        v_id_ped := v_id_ped + 1;
    END LOOP;
    DBMS_OUTPUT.PUT_LINE('Pedidos insertados con WHILE: ' || v_insertados);
END;

Fase BUPDATE dentro de FOR (Procesamiento por ID)

Objetivo: Simular un reabastecimiento crítico. Aumentar en 2 unidades el stock para productos entre ID 10 al 40 (avanzando de 10 en 10) que tengan stock menor o igual a 5.

DECLARE
    v_id_prod NUMBER;
    v_reabastecidos NUMBER := 0;
BEGIN
    FOR i IN 1..4 LOOP
        v_id_prod := i * 10; -- Genera 10, 20, 30, 40
        
        UPDATE productos
        SET stock = stock + 2
        WHERE id_producto = v_id_prod AND stock <= 5;
        
        DBMS_OUTPUT.PUT_LINE('Revisado producto ID ' || v_id_prod || ' para posible reabastecimiento.');
        v_reabastecidos := v_reabastecidos + 1;
    END LOOP;
    DBMS_OUTPUT.PUT_LINE('Total iteraciones de reabastecimiento: ' || v_reabastecidos);
END;
/

Fase CUPDATE dentro de WHILE (Cambio de estado)

Objetivo: Subir gradualmente a un cliente de regular a empresa. Consultar el estado del cliente ID 2. Usar un bucle WHILE que corra hasta que sea 'EMPRESA', ascendiendo a 'VIP' y luego a 'EMPRESA' con UPDATEs y recargando la variable de control.

DECLARE
    v_tipo clientes.tipo_cliente%TYPE;
BEGIN
    -- Obtenemos el tipo inicial (REGULAR)
    SELECT tipo_cliente INTO v_tipo FROM clientes WHERE id_cliente = 2;
    
    WHILE v_tipo <> 'EMPRESA' LOOP
        IF v_tipo = 'REGULAR' THEN
            UPDATE clientes SET tipo_cliente = 'VIP' WHERE id_cliente = 2;
            DBMS_OUTPUT.PUT_LINE('Ascendido a VIP.');
        ELSIF v_tipo = 'VIP' THEN
            UPDATE clientes SET tipo_cliente = 'EMPRESA' WHERE id_cliente = 2;
            DBMS_OUTPUT.PUT_LINE('Ascendido a EMPRESA.');
        END IF;
        
        -- Volvemos a leer la BD para actualizar la variable del bucle
        SELECT tipo_cliente INTO v_tipo FROM clientes WHERE id_cliente = 2;
    END LOOP;
END;
/
Verificación con SELECT: SELECT * FROM clientes WHERE id_cliente = 2;

Módulo 3: Ejercicio Integrador (Desafío Combinado)

Escenario de Negocio

La gerencia de la Tienda necesita realizar un proceso diario automático:

  1. **Reabastecimiento condicional**: Consultar el stock actual de un producto crítico ("Mouse Gamer", ID 10) usando SELECT INTO. Si el stock es inferior a 20, sumarle 5 unidades. De lo contrario, restarle 2 unidades por regularización de mermas (UPDATE).
  2. **Campaña de Logística**: Para registrar logs de la actividad, se registrarán en la tabla clientes 3 usuarios ficticios de auditoría (IDs del 210 al 212) con nombre 'Audit_USER_X' y correo 'auditX@tienda.com' usando un bucle FOR (INSERT).
  3. **Limpieza de registros obsoletos**: Eliminar el cliente con ID 200 (si quedó de pruebas anteriores) para evitar acumulación (DELETE).

1. Recuerda usar SELECT stock INTO v_stock FROM productos WHERE id_producto = 10; para obtener el stock inicial.

2. Usa una estructura IF v_stock < 20 THEN ... ELSE ... END IF; para aplicar el UPDATE correspondiente.

3. En el bucle de logs, usa FOR i IN 210..212 LOOP y concatena el iterador `i` en las cadenas de inserción. 

DECLARE
    v_stock productos.stock%TYPE;
    v_afectados NUMBER := 0;
BEGIN
    -- 1. Consultar el stock actual del Mouse Gamer
    SELECT stock INTO v_stock 
    FROM productos 
    WHERE id_producto = 10;
    
    -- 2. Decisión lógica del DML 1 (UPDATE)
    IF v_stock < 20 THEN
        UPDATE productos 
        SET stock = stock + 5 
        WHERE id_producto = 10;
        DBMS_OUTPUT.PUT_LINE('Stock crítico: Reabastecidas 5 unidades.');
    ELSE
        UPDATE productos 
        SET stock = stock - 2 
        WHERE id_producto = 10;
        DBMS_OUTPUT.PUT_LINE('Stock abundante: Ajustadas 2 unidades por merma.');
    END IF;
    
    -- 3. Bucle FOR del DML 2 (INSERT masivo de Auditorías)
    FOR i IN 210..212 LOOP
        INSERT INTO clientes (id_cliente, nombre, correo, tipo_cliente)
        VALUES (i, 'Audit_USER_' || (i - 209), 'audit' || (i - 209) || '@tienda.com', 'REGULAR');
        
        v_afectados := v_afectados + 1;
    END LOOP;
    DBMS_OUTPUT.PUT_LINE('Logs creados en clientes: ' || v_afectados);
    
    -- 4. DML 3 (DELETE de limpieza)
    DELETE FROM clientes 
    WHERE id_cliente = 200;
    
    DBMS_OUTPUT.PUT_LINE('Proceso de limpieza de cliente 200 ejecutado.');
    
    COMMIT;
END;
/
Consultas de Verificación:

SELECT * FROM clientes WHERE id_cliente BETWEEN 210 AND 212;

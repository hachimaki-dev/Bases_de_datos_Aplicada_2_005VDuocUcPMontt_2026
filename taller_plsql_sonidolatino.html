Taller PL/SQL: SonidoLatino

DML Progresivo con Control de Flujo

Caso de Negocio

SonidoLatino es una plataforma de streaming especializada en música chilena, andina, pop de los 80 y rock de los 70, 80 y 90. El equipo de datos necesita automatizar tareas de mantenimiento, promociones y reportes mediante PL/SQL. Tu rol es implementar los bloques que resuelvan cada requerimiento.

Estructura de la Base de Datos

artistas id_artista, nombre, pais, genero, activo
canciones id_cancion, titulo, id_artista, anio, duracion_seg, reproducciones, precio_licencia
usuarios id_usuario, nombre, correo, plan, creditos
reproducciones id_repro, id_usuario, id_cancion, fecha, completada

Preparación del Entorno (Script de Base de Datos)

Copia y ejecuta el siguiente script en tu cliente Oracle SQL Developer para inicializar la base de datos de SonidoLatino.

setup_sonidolatino.sql 
-- Limpieza
DROP TABLE reproducciones CASCADE CONSTRAINTS PURGE;
DROP TABLE canciones     CASCADE CONSTRAINTS PURGE;
DROP TABLE artistas      CASCADE CONSTRAINTS PURGE;
DROP TABLE usuarios      CASCADE CONSTRAINTS PURGE;

CREATE TABLE artistas (
    id_artista  NUMBER PRIMARY KEY,
    nombre      VARCHAR2(100) NOT NULL,
    pais        VARCHAR2(60),
    genero      VARCHAR2(40),
    activo      NUMBER(1) DEFAULT 1,
    CHECK (activo IN (0,1))
);

CREATE TABLE canciones (
    id_cancion       NUMBER PRIMARY KEY,
    titulo           VARCHAR2(150) NOT NULL,
    id_artista       NUMBER NOT NULL,
    anio             NUMBER(4),
    duracion_seg     NUMBER NOT NULL,
    reproducciones   NUMBER DEFAULT 0,
    precio_licencia  NUMBER(10,2) DEFAULT 0,
    FOREIGN KEY (id_artista) REFERENCES artistas(id_artista),
    CHECK (duracion_seg > 0),
    CHECK (reproducciones >= 0),
    CHECK (precio_licencia >= 0)
);

CREATE TABLE usuarios (
    id_usuario  NUMBER PRIMARY KEY,
    nombre      VARCHAR2(100) NOT NULL,
    correo      VARCHAR2(100) UNIQUE,
    plan        VARCHAR2(20) DEFAULT 'GRATIS',
    creditos    NUMBER DEFAULT 0,
    CHECK (plan IN ('GRATIS','BASICO','PREMIUM'))
);

CREATE TABLE reproducciones (
    id_repro    NUMBER PRIMARY KEY,
    id_usuario  NUMBER NOT NULL,
    id_cancion  NUMBER NOT NULL,
    fecha       DATE NOT NULL,
    completada  NUMBER(1) DEFAULT 1,
    FOREIGN KEY (id_usuario) REFERENCES usuarios(id_usuario),
    FOREIGN KEY (id_cancion) REFERENCES canciones(id_cancion),
    CHECK (completada IN (0,1))
);

Carga Masiva de Canciones Demo

INSERT + FOR Inicial

OBJETIVO: Cargar 4 canciones de prueba (IDs 501–504) asociadas al artista 500. Todas con duración de 200 segundos, 0 reproducciones y precio_licencia de 1000. El título de cada canción debe tener el formato "Demo Track 1", "Demo Track 2", etc.

Pista: Usa un ciclo FOR i IN 1..4. El ID de la canción se calcula como 500 + i, y el título se construye concatenando texto con el operador ||.

solucion_r2.sql 
DECLARE
    v_insertadas NUMBER := 0;
BEGIN
    FOR i IN 1..4 LOOP
        INSERT INTO canciones
            (id_cancion, titulo, id_artista, anio, duracion_seg, reproducciones, precio_licencia)
        VALUES
            (500 + i, 'Demo Track ' || i, 500, 2026, 200, 0, 1000);
        v_insertadas := v_insertadas + 1;
    END LOOP;
    DBMS_OUTPUT.PUT_LINE('Canciones demo insertadas: ' || v_insertadas);
END;

Descuento de Licencia

solucion_r3.sql 
DECLARE
    v_precio_actual  canciones.precio_licencia%TYPE;
    v_precio_nuevo   NUMBER(10,2);
BEGIN
    SELECT precio_licencia INTO v_precio_actual
    FROM canciones WHERE id_cancion = 50;

    v_precio_nuevo := v_precio_actual * 0.85;

    UPDATE canciones
    SET precio_licencia = v_precio_nuevo
    WHERE id_cancion = 50;

    DBMS_OUTPUT.PUT_LINE('Anterior: ' || v_precio_actual || ' | Nuevo: ' || v_precio_nuevo);
END;

Desafío R4

Reajuste Masivo de Licencias

UPDATE + FOR + IF Intermedia

OBJETIVO: Recorrer las canciones con IDs 10, 30, 50, 70 y 90 (avanzando de 20 en 20). Si su precio_licencia es igual a 0, actualizarlo a 3000. Imprimir en pantalla el ID procesado y la acción realizada en cada iteración.

Pista: Diseña un ciclo FOR i IN 1..5 y calcula el ID correspondiente usando la fórmula i * 20. Usa condicionales IF para validar el precio antes de lanzar el UPDATE.

Código de solución protegido

Desbloquéalo arriba ingresando la contraseña del taller.

solucion_r4.sql 
DECLARE
    v_id_cancion NUMBER;
    v_precio     canciones.precio_licencia%TYPE;
BEGIN
    FOR i IN 1..5 LOOP
        v_id_cancion := i * 20;

        SELECT precio_licencia INTO v_precio
        FROM canciones WHERE id_cancion = v_id_cancion;

        IF v_precio = 0 THEN
            UPDATE canciones
            SET precio_licencia = 3000
            WHERE id_cancion = v_id_cancion;
            DBMS_OUTPUT.PUT_LINE('ID ' || v_id_cancion || ': licencia asignada a 3000.');
        ELSE
            DBMS_OUTPUT.PUT_LINE('ID ' || v_id_cancion || ': ya tiene precio (' || v_precio || ').');
        END IF;
    END LOOP;
END;
/
Consulta de Verificación: SELECT id_cancion, titulo, precio_licencia FROM canciones WHERE id_cancion IN (10,30,50,70,90);

Desafío R5

Ascenso Automático de Plan de Usuario

UPDATE + WHILE Intermedia

OBJETIVO: Promover gradualmente el plan del usuario ID 3 (Isidora, actualmente plan 'GRATIS'). Diseñar un ciclo WHILE que ascienda al usuario secuencialmente en el orden 'GRATIS' → 'BASICO' → 'PREMIUM', leyendo el plan directamente de la base de datos en cada iteración del bucle.

Pista: Lee el plan inicial antes de entrar al bucle. Dentro del WHILE, aplica el UPDATE y vuelve a ejecutar la consulta SELECT plan INTO ... al final de la iteración para evaluar la condición de parada del bucle.

Código de solución protegido

Desbloquéalo arriba ingresando la contraseña del taller.

solucion_r5.sql 
DECLARE
    v_plan usuarios.plan%TYPE;
BEGIN
    SELECT plan INTO v_plan FROM usuarios WHERE id_usuario = 3;

    WHILE v_plan <> 'PREMIUM' LOOP
        IF v_plan = 'GRATIS' THEN
            UPDATE usuarios SET plan = 'BASICO' WHERE id_usuario = 3;
            DBMS_OUTPUT.PUT_LINE('Ascendido a BASICO.');
        ELSIF v_plan = 'BASICO' THEN
            UPDATE usuarios SET plan = 'PREMIUM' WHERE id_usuario = 3;
            DBMS_OUTPUT.PUT_LINE('Ascendido a PREMIUM.');
        END IF;
        SELECT plan INTO v_plan FROM usuarios WHERE id_usuario = 3;
    END LOOP;
    DBMS_OUTPUT.PUT_LINE('Plan final: ' || v_plan);
END;
/
Consulta de Verificación: SELECT id_usuario, nombre, plan FROM usuarios WHERE id_usuario = 3;

Preguntas de discusión en parejas:

  • ¿Por qué es fundamental volver a consultar la tabla de la BD antes de finalizar cada ciclo en vez de simplemente cambiar el valor de la variable en memoria?
  • ¿Cuántas iteraciones del bucle se ejecutarían si el usuario originalmente ya tuviera el plan 'PREMIUM'?

Simulación de Reproducciones Masivas

Pista: Declara un contador de IDs que se inicie en 500. La condición del bucle será v_id_repro <= 503. No olvides incrementar el ID en cada paso para evitar un bucle sin fin.

Desbloquéalo arriba ingresando la contraseña del taller.

solucion_r6.sql 
DECLARE
    v_id_repro NUMBER := 500;
    v_total    NUMBER := 0;
BEGIN
    WHILE v_id_repro <= 503 LOOP
        INSERT INTO reproducciones (id_repro, id_usuario, id_cancion, fecha, completada)
        VALUES (v_id_repro, 1, 10, SYSDATE, 1);

        v_total    := v_total + 1;
        v_id_repro := v_id_repro + 1;
    END LOOP;
    DBMS_OUTPUT.PUT_LINE('Reproducciones simuladas: ' || v_total);
END;

Créditos por Fidelidad

UPDATE + FOR + IF Intermedia

OBJETIVO: Recorrer de forma secuencial los usuarios con IDs del 1 al 4. Si su plan es 'PREMIUM', sumar 100 créditos. Si su plan es 'BASICO', sumar 25 créditos. Si es 'GRATIS', no realizar modificaciones. Imprimir por pantalla el nombre y la acción de reajuste realizada.

Pista: Emplea un bucle FOR i IN 1..4. En cada pasada, lee el nombre y el plan actual, y luego determina el reajuste en créditos utilizando estructuras condicionales IF-ELSIF-ELSE.

Código de solución protegido

Desbloquéalo arriba ingresando la contraseña del taller.

solucion_r8.sql 
DECLARE
    v_plan   usuarios.plan%TYPE;
    v_nombre usuarios.nombre%TYPE;
BEGIN
    FOR i IN 1..4 LOOP
        SELECT nombre, plan INTO v_nombre, v_plan
        FROM usuarios WHERE id_usuario = i;

        IF v_plan = 'PREMIUM' THEN
            UPDATE usuarios SET creditos = creditos + 100 WHERE id_usuario = i;
            DBMS_OUTPUT.PUT_LINE(v_nombre || ' (PREMIUM): +100 creditos.');
        ELSIF v_plan = 'BASICO' THEN
            UPDATE usuarios SET creditos = creditos + 25  WHERE id_usuario = i;
            DBMS_OUTPUT.PUT_LINE(v_nombre || ' (BASICO): +25 creditos.');
        ELSE
            DBMS_OUTPUT.PUT_LINE(v_nombre || ' (GRATIS): sin cambios.');
        END IF;
    END LOOP;
END;
/
Consulta de Verificación: SELECT id_usuario, nombre, plan, creditos FROM usuarios;

Desafío R10

Desactivar Artistas sin Licencias Activas

UPDATE + WHILE + IF Avanzada

OBJETIVO: Recorrer secuencialmente los artistas con IDs 60, 70 y 80 en incrementos de 10 usando un ciclo WHILE. Si el artista no posee ninguna canción asociada que tenga un precio_licencia mayor a 0, actualizar su estado a inactivo (activo = 0). Imprimir el nombre del artista y la acción realizada.

Pista: Inicializa el contador del ciclo en 60. Para cada artista, realiza un COUNT(*) de sus canciones con precio_licencia > 0. Utiliza condicionales para determinar si ejecutas la desactivación.

Código de solución protegido

Desbloquéalo arriba ingresando la contraseña del taller.

solucion_r10.sql 
DECLARE
    v_id     NUMBER := 60;
    v_nombre artistas.nombre%TYPE;
    v_cnt    NUMBER;
BEGIN
    WHILE v_id <= 80 LOOP
        SELECT nombre INTO v_nombre
        FROM artistas WHERE id_artista = v_id;

        SELECT COUNT(*) INTO v_cnt
        FROM canciones
        WHERE id_artista = v_id AND precio_licencia > 0;

        IF v_cnt = 0 THEN
            UPDATE artistas SET activo = 0 WHERE id_artista = v_id;
            DBMS_OUTPUT.PUT_LINE(v_nombre || ': desactivado (sin licencias activas).');
        ELSE
            DBMS_OUTPUT.PUT_LINE(v_nombre || ': activo (' || v_cnt || ' canciones con licencia).');
        END IF;

        v_id := v_id + 10;
    END LOOP;
END;
/
Consulta de Verificación: SELECT id_artista, nombre, activo FROM artistas WHERE id_artista IN (60,70,80);

Preguntas de discusión en parejas:

  • ¿Por qué se prefiere un cambio de estado en una columna lógica (soft delete) en lugar de borrar por completo la fila usando DELETE?
  • ¿Qué problema surgiría si alguno de los IDs del ciclo no existiera dentro de la tabla de artistas?
